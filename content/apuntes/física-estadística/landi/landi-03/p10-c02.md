$$\begin{aligned}
G_Y(s) &= G_{X_1}(s) \cdots G_{X_m}(s)\\
&= e^{is\mu_1-\frac{s^2\sigma_1^2}{2}}e^{is\mu_2-\frac{s^2\sigma_2^2}{2}}\cdots e^{is\mu_m-\frac{s^2\sigma_m^2}{2}}\\
&= \exp\left\{is(\mu_1+\cdots+\mu_m)-\frac{s^2}{2}(\sigma_1^2+\cdots+\sigma_m^2)\right\}
\end{aligned}\tag{21}$$
