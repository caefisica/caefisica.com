$$\begin{aligned}
z &= \sum_m e^{-\beta E_m} = 1 + e^{-\beta(E+\Delta)} + e^{-\beta(E-\Delta)}\\
&= 1 + e^{-\beta E}\underbrace{\left(e^{-\beta\Delta}+e^{\beta\Delta}\right)}_{2\cosh\beta\Delta}\\
