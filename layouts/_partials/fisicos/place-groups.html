{{- /* Returns who studied or worked at one place of data/physicists.json: Pregrado, Posgrado, Trabajan and Trabajaron, each a list of dict person and note, with n, the count of distinct people. People keep the data's order, by surname. Context: the place. */ -}} {{- $place := . -}} {{- $undergrad := slice "Bachillerato" "Licenciatura" -}} {{- $under := slice -}} {{- $grad := slice -}} {{- $works := slice -}} {{- $worked := slice -}} {{- $names := slice -}} {{- range hugo.Data.physicists.people -}} {{- $person := . -}} {{- $u := slice -}} {{- $g := slice -}} {{- range .studied -}} {{- if eq .place $place.id -}} {{- $text := .level -}} {{- with .field }}{{ if ne . $text }}{{ $text = printf "%s (%s)" $text (lower .) }}{{ end }}{{ end -}} {{- if in $undergrad .level }}{{ $u = $u | append $text }}{{ else }}{{ $g = $g | append $text }}{{ end -}} {{- end -}} {{- end -}} {{- with $u }}{{ $under = $under | append (dict "person" $person "note" (delimit . " · ")) }}{{ end -}} {{- with $g }}{{ $grad = $grad | append (dict "person" $person "note" (delimit . " · ")) }}{{ end -}} {{- if or $u $g }}{{ $names = $names | append $person.slug }}{{ end -}} {{- range .works -}} {{- if eq .place $place.id -}} {{- $works = $works | append (dict "person" $person "note" .role) -}} {{- $names = $names | append $person.slug -}} {{- end -}} {{- end -}} {{- /* Someone who works there now is under Trabajan, so their earlier posts there are not repeated. */ -}} {{- if not (where .works "place" $place.id) -}} {{- $roles := slice -}} {{- range where .jobs "place" $place.id }}{{ $roles = $roles | append .role }}{{ end -}} {{- with $roles }} {{- $worked = $worked | append (dict "person" $person "note" (delimit (uniq .) " · ")) -}} {{- $names = $names | append $person.slug -}} {{- end -}} {{- end -}} {{- end -}} {{- $groups := slice (dict "title" "Pregrado" "rows" $under) (dict "title" "Posgrado" "rows" $grad) (dict "title" "Trabajan" "rows" $works) (dict "title" "Trabajaron" "rows" $worked) -}} {{- return dict "groups" $groups "n" ($names | uniq | len) -}}
