{{- /* Renders footnote marks for a list of source keys. Context: dict with page and keys. Each mark links its entry in the page's reference list. The link holds a card with the source's title and site, shown on hover and focus, so a reader checks a source without leaving the line; the card's text also names the link for a screen reader. */ -}} {{- $sources := .page.Params.sources -}} {{- range .keys }} {{- $key := . }} {{- $n := 0 }} {{- $source := dict }} {{- range $i, $s := $sources }}{{ if eq $s.key $key }}{{ $n = add $i 1 }}{{ $source = $s }}{{ end }}{{ end }} {{- if not $n }}{{ errorf "%s: no source with key %q" $.page.File.Path $key }}{{ end -}} {{- $n -}} {{ $source.title }} {{ strings.TrimPrefix "www." (urls.Parse $source.url).Host }} {{- end -}} {{- /**/ -}}
