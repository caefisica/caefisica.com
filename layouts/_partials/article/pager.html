{{- /* Links to the previous and next sibling, in the order the section tree shows. A top-level section has no siblings: the site's sections are not a reading order. Context: the page. */ -}} {{- $page := . -}} {{- $prev := false -}} {{- $next := false -}} {{- with .Parent -}} {{ if not .IsHome -}} {{- $siblings := partial "docs/children.html" . -}} {{- range $i, $p := $siblings -}} {{- if $p.Eq $page -}} {{- if gt $i 0 }}{{ $prev = index $siblings (sub $i 1) }}{{ end -}} {{- if lt (add $i 1) (len $siblings) }}{{ $next = index $siblings (add $i 1) }}{{ end -}} {{- end -}} {{- end -}} {{- end }} {{ end -}} {{- if or $prev $next -}}
{{- with $prev }} Anterior {{ .LinkTitle }} {{- end }} {{- with $next }} Siguiente {{ .LinkTitle }} {{- end }}
{{- end -}}
