{{- /* Computes responsive image data from a Hugo image resource. Input: dict "image" $imageResource Returns: dict with keys "src", "srcset" src — the first width in params, for browsers that ignore srcset srcset — comma-joined srcset string */ -}} {{- $image := .image -}} {{- $imgSrc := "" -}} {{- $imgSrcSet := slice -}} {{- $widths := site.Params.landscapePhotoWidths -}} {{- if gt $image.Height $image.Width -}} {{- $widths = site.Params.portraitPhotoWidths -}} {{- end -}} {{- range $widths -}} {{- $srcUrl := (printf "%dx" . | $image.Resize).RelPermalink -}} {{- if eq $imgSrc "" -}}{{- $imgSrc = $srcUrl -}}{{- end -}} {{- $imgSrcSet = $imgSrcSet | append (printf "%s %dw" $srcUrl .) -}} {{- end -}} {{- return (dict "src" $imgSrc "srcset" (delimit $imgSrcSet ",")) -}}
